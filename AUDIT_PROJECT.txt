module.exports = async function handler(req, res) {
  res.setHeader("Cache-Control", "no-store");

  if (req.method !== "POST") {
    res.setHeader("Allow", "POST");
    return res.status(405).json({ error: "Method not allowed" });
  }

  const supabaseUrl = process.env.SUPABASE_URL;
  const serviceKey = process.env.SUPABASE_SERVICE_ROLE_KEY;

  if (!supabaseUrl || !serviceKey) {
    return res.status(500).json({ error: "Konfigurasi server belum lengkap" });
  }

  try {
    const b = req.body || {};
    const command = {
      mode: b.mode,
      power: b.power,
      temp: Number(b.temp),
      ac_mode: b.ac_mode,
      fan: b.fan,
      swing: b.swing,
      quiet: b.quiet,
      turbo: b.turbo,
      sleep: b.sleep
    };

    if (!["manual", "auto"].includes(command.mode))
      return res.status(400).json({ error: "Mode kontrol tidak valid" });

    if (typeof command.power !== "boolean")
      return res.status(400).json({ error: "Power tidak valid" });

    if (!Number.isInteger(command.temp) || command.temp < 16 || command.temp > 30)
      return res.status(400).json({ error: "Suhu harus 16-30°C" });

    if (!["cool", "dry", "fan", "heat"].includes(command.ac_mode))
      return res.status(400).json({ error: "Mode AC tidak valid" });

    if (!["auto", "low", "med", "medium", "high"].includes(command.fan))
      return res.status(400).json({ error: "Kipas tidak valid" });

    for (const key of ["swing", "quiet", "turbo", "sleep"]) {
      if (typeof command[key] !== "boolean")
        return res.status(400).json({ error: "Pengaturan " + key + " tidak valid" });
    }

    const response = await fetch(
      supabaseUrl.replace(/\/+$/, "") + "/rest/v1/rpc/set_ac_command",
      {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          "apikey": serviceKey,
          "Authorization": "Bearer " + serviceKey
        },
        body: JSON.stringify({ p_command: command })
      }
    );

    const result = await response.json().catch(() => null);

    if (!response.ok) {
      console.error("Supabase RPC error:", response.status, result);
      return res.status(502).json({ error: (result && (result.message || result.details || result.hint)) || "Gagal menyimpan perintah ke Supabase", code: result && result.code });
    }

    return res.status(200).json(result);
  } catch (error) {
    console.error(error);
    return res.status(500).json({ error: "Kesalahan server" });
  }
};

module.exports = async function handler(req, res) {
  res.setHeader("Cache-Control", "no-store");

  if (req.method !== "GET") {
    res.setHeader("Allow", "GET");
    return res.status(405).json({ error: "Method not allowed" });
  }

  const base = process.env.SUPABASE_URL;
  const key = process.env.SUPABASE_SERVICE_ROLE_KEY;

  if (!base || !key) {
    return res.status(500).json({
      error: "Konfigurasi Supabase server belum lengkap"
    });
  }

  try {
    // Menambahkan est_kwh_bulan_ini dan est_biaya_bulan_ini ke dalam select
    const url = base.replace(/\/+$/, "") +
      "/rest/v1/ac_state?device_id=eq.esp32-kamar&select=device_id,rev,power,temp,ac_temp_command,ac_mode,fan,swing,quiet,turbo,sleep,mode,est_kwh_bulan_ini,est_biaya_bulan_ini,updated_by,updated_at&limit=1";

    const response = await fetch(url, {
      headers: {
        apikey: key,
        Authorization: "Bearer " + key
      },
      cache: "no-store"
    });

    if (!response.ok) {
      console.error("Supabase:", response.status, await response.text());
      return res.status(502).json({ error: "Gagal membaca status AC" });
    }

    const rows = await response.json();
    const state = rows[0] || null;
    const lastSeenMs = state && state.updated_at ? Date.parse(state.updated_at) : NaN;
    const ageMs = Date.now() - lastSeenMs;
    const online = Number.isFinite(lastSeenMs) && ageMs >= -5000 && ageMs <= 45000;
    
    return res.status(200).json({
      state,
      device: {
        online,
        last_seen: state && state.updated_at ? state.updated_at : null
      }
    });
  } catch (error) {
    console.error(error);
    return res.status(500).json({ error: "Kesalahan server" });
  }
};

import { createClient } from "npm:@supabase/supabase-js@2";

const SUPABASE_URL = Deno.env.get("SUPABASE_URL")!;
const SERVICE_ROLE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!;
const DEVICE_TOKEN = Deno.env.get("DEVICE_TOKEN");

if (!SUPABASE_URL || !SERVICE_ROLE_KEY || !DEVICE_TOKEN) {
  throw new Error("Secret SUPABASE_URL, SUPABASE_SERVICE_ROLE_KEY, atau DEVICE_TOKEN belum tersedia");
}

const db = createClient(SUPABASE_URL, SERVICE_ROLE_KEY);

const cors = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Methods": "POST, OPTIONS",
  "Access-Control-Allow-Headers": "authorization, content-type, x-device-token, apikey",
  "Content-Type": "application/json",
};

const FIELDS = [
  "power",
  "temp",
  "ac_temp_command",
  "ac_mode",
  "fan",
  "swing",
  "quiet",
  "turbo",
  "sleep",
  "mode",
];

function json(body: unknown, status = 200) {
  return new Response(JSON.stringify(body), { status, headers: cors });
}

function errorMessage(error: unknown) {
  return error instanceof Error ? error.message : String(error);
}

function validDeviceId(value: unknown): value is string {
  return typeof value === "string" && value.trim().length > 0 && value.length <= 100;
}

function validState(state: any) {
  if (!state || typeof state !== "object" || Array.isArray(state)) return false;

  if (state.temp !== undefined && (!Number.isInteger(state.temp) || state.temp < 16 || state.temp > 30)) return false;
  if (state.power !== undefined && state.power !== null && typeof state.power !== "boolean") return false;

  for (const key of ["swing", "quiet", "turbo", "sleep"]) {
    if (state[key] !== undefined && typeof state[key] !== "boolean") return false;
  }

  if (state.ac_mode !== undefined && !["cool", "dry", "fan", "heat"].includes(state.ac_mode)) return false;
  if (state.fan !== undefined && !["auto", "low", "med", "medium", "high"].includes(state.fan)) return false;
  if (state.mode !== undefined && !["manual", "auto"].includes(state.mode)) return false;

  return true;
}

async function merge(deviceId: string, patch: any, by: string, baseRev?: number) {
  if (!validDeviceId(deviceId)) throw new Error("device_id tidak valid");
  if (!validState(patch)) throw new Error("Format state tidak valid");

  const { data: current, error: readError } = await db.from("ac_state").select("*").eq("device_id", deviceId).maybeSingle();
  if (readError) throw readError;

  const currentRev = Number(current?.rev ?? 0);
  if (baseRev !== undefined && baseRev !== currentRev) {
    return { conflict: true, rev: currentRev, state: current };
  }

  const update: Record<string, unknown> = {
    rev: currentRev + 1,
    updated_by: by,
    updated_at: new Date().toISOString(),
  };

  for (const key of FIELDS) {
    if (Object.prototype.hasOwnProperty.call(patch, key)) {
      update[key] = patch[key];
    }
  }

  let result;
  if (current) {
    const { data, error } = await db.from("ac_state").update(update).eq("device_id", deviceId).eq("rev", currentRev).select().maybeSingle();
    if (error) throw error;
    result = data;
  } else {
    const { data, error } = await db.from("ac_state").insert({
      device_id: deviceId,
      ...update,
      ...Object.fromEntries(FIELDS.filter((key) => Object.prototype.hasOwnProperty.call(patch, key)).map((key) => [key, patch[key]])),
    }).select().single();
    if (error) throw error;
    result = data;
  }

  if (!result) {
    const { data: latest, error } = await db.from("ac_state").select("*").eq("device_id", deviceId).maybeSingle();
    if (error) throw error;
    return { conflict: true, rev: Number(latest?.rev ?? 0), state: latest };
  }

  return { conflict: false, rev: result.rev, state: result };
}

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") return new Response("ok", { headers: cors });
  if (req.method !== "POST") return json({ success: false, error: "Gunakan POST" }, 405);

  let body: any;
  try { body = await req.json(); } catch { return json({ success: false, error: "JSON request tidak valid" }, 400); }

  try {
    const deviceAction = ["sensor", "get_state", "report_state"].includes(body.action);
    const deviceAuthorized = Boolean(DEVICE_TOKEN) && req.headers.get("x-device-token") === DEVICE_TOKEN;

    if (deviceAction && !deviceAuthorized) return json({ success: false, error: "Token perangkat tidak valid" }, 401);
    if (deviceAction && !validDeviceId(body.device_id)) return json({ success: false, error: "device_id tidak valid" }, 400);

    if (body.action === "sensor") {
      const suhu = body.suhu;
      const kelembapan = body.kelembapan;
      if (typeof suhu !== "number" || typeof kelembapan !== "number" || !Number.isFinite(suhu) || !Number.isFinite(kelembapan)) {
        return json({ success: false, error: "Nilai suhu/kelembapan tidak valid" }, 400);
      }

      const row = {
        device_id: body.device_id,
        suhu,
        kelembapan,
        control_mode: ["manual", "auto"].includes(body.control_mode) ? body.control_mode : "manual",
        desired_power: body.desired_power === true ? "true" : body.desired_power === false ? "false" : "unknown",
        ac_temp_command: body.ac_temp_command ?? null,
        ac_mode_command: body.ac_mode_command ?? null,
        last_action: body.last_action ?? null,
      };

      const { data, error } = await db.from("sensor_readings").insert(row).select("id, device_id, suhu, kelembapan, recorded_at").single();
      if (error) return json({ success: false, error: "Gagal menyimpan sensor", details: error.message }, 500);

      return json({ success: true, message: "Data sensor tersimpan", data });
    }

    if (body.action === "get_state") {
      const { data, error } = await db.from("ac_state").select("*").eq("device_id", body.device_id).maybeSingle();
      if (error) throw error;
      const sinceRev = Number(body.since_rev ?? 0);
      return json({ state: data && Number(data.rev) > sinceRev ? data : null });
    }

    if (body.action === "report_state") {
      if (!body.state || typeof body.state !== "object") return json({ success: false, error: "State tidak valid" }, 400);
      const result = await merge(body.device_id, body.state, "device", Number(body.base_rev ?? 0));
      return json(result);
    }

    if (body.action === "set_state") {
      const authorization = req.headers.get("Authorization");
      if (!authorization?.startsWith("Bearer ")) return json({ success: false, error: "Login diperlukan" }, 401);

      const userClient = createClient(SUPABASE_URL, Deno.env.get("SUPABASE_ANON_KEY")!, { global: { headers: { Authorization: authorization } } });
      const { data: userData, error: userError } = await userClient.auth.getUser();
      if (userError || !userData.user) return json({ success: false, error: "Sesi login tidak valid" }, 401);

      if (!validDeviceId(body.device_id) || !validState(body.state)) return json({ success: false, error: "Device ID atau state tidak valid" }, 400);

      const result = await merge(body.device_id, body.state, "web", body.base_rev === undefined ? undefined : Number(body.base_rev));
      return json(result);
    }

    return json({ success: false, error: "Action tidak dikenal" }, 400);
  } catch (error) {
    return json({ success: false, error: "Operasi database gagal", details: errorMessage(error) }, 500);
  }
});

const jakartaFormatter = new Intl.DateTimeFormat("en-CA", {
  timeZone: "Asia/Jakarta",
  year: "numeric",
  month: "2-digit",
  day: "2-digit"
});

function jakartaDateString(date = new Date()) {
  const parts = jakartaFormatter.formatToParts(date);
  const values = {};

  for (const part of parts) {
    if (part.type !== "literal") values[part.type] = part.value;
  }

  return `${values.year}-${values.month}-${values.day}`;
}

function jakartaHour(date) {
  return Number(new Intl.DateTimeFormat("en-GB", {
    timeZone: "Asia/Jakarta",
    hour: "2-digit",
    hourCycle: "h23"
  }).format(date));
}

function normalizeRecord(item) {
  if (!item) return null;

  const suhu = Number(item.suhu);
  const kelembapan = Number(item.kelembapan);

async function fetchTodayHistory() {
  const today = jakartaDateString();
  const start = `${today}T00:00:00+07:00`;

  const tomorrowDate = new Date(
    new Date(start).getTime() + 24 * 60 * 60 * 1000
  );

  const tomorrow = jakartaDateString(tomorrowDate);
  const end = `${tomorrow}T00:00:00+07:00`;
  const allRows = [];
  let offset = 0;

  while (true) {
    const query = new URLSearchParams({
      select: "suhu,kelembapan,recorded_at",
      and: `(recorded_at.gte.${start},recorded_at.lt.${end})`,
      order: "recorded_at.asc",
      limit: String(PAGE_SIZE),
      offset: String(offset)
    });

    const rows = await supabaseFetch(
      `${TABLE_URL}?${query.toString()}`
    );

    allRows.push(...rows);

    if (rows.length < PAGE_SIZE) break;
    offset += rows.length;
  }

  return allRows.map(normalizeRecord).filter(Boolean);
}

function analyzeByHour(records) {
  const today = jakartaDateString();
  const now = new Date();
  const currentHour = jakartaHour(now);
  const elapsedSeconds = now.getMinutes() * 60 + now.getSeconds();

  // ESP32 direncanakan mengirim data setiap 30 detik.
  const TARGET_PER_HOUR = 120;
  const MIN_SAMPLES = 60;
  const MIN_COVERAGE = 0.75;

  const groups = new Map();

  for (const record of records) {
    if (!record || !record.date || !Number.isFinite(record.suhu)) continue;
    if (jakartaDateString(record.date) !== today) continue;

    const hour = jakartaHour(record.date);
    if (!groups.has(hour)) groups.set(hour, []);
    groups.get(hour).push(record);
  }

  const hourly = [];

  for (const [hour, values] of groups.entries()) {
    // Jam lampau memakai target satu jam penuh.
    // Jam berjalan memakai target sesuai durasi yang telah berlalu.
    const expected = hour === currentHour
      ? Math.min(TARGET_PER_HOUR, Math.ceil(elapsedSeconds / 30))
      : TARGET_PER_HOUR;

    const temps = values.map(item => item.suhu);
    const count = temps.length;
    const coverage = expected > 0 ? Math.min(1, count / expected) : 0;

    if (count < MIN_SAMPLES || coverage < MIN_COVERAGE) continue;

    hourly.push({
      hour,
      average: temps.reduce((sum, value) => sum + value, 0) / count,
      min: Math.min(...temps),
      max: Math.max(...temps),
      count,
      expected,
      coverage
    });
  }

  hourly.sort((a, b) => a.hour - b.hour);

  const hotHour = $("hotHour");
  const coolHour = $("coolHour");
  const hotDetail = $("hotDetail");
  const coolDetail = $("coolDetail");
  const insight = $("insightText");

  if (!hotHour || !coolHour || !hotDetail || !coolDetail || !insight) {
    console.error("Elemen analisis suhu tidak ditemukan di HTML.");
    return;
  }

  const range = hour =>
    `${String(hour).padStart(2, "0")}.00–${String((hour + 1) % 24).padStart(2, "0")}.00`;

  function showResult(element, detail, item) {
    element.textContent = range(item.hour);
    detail.textContent =
      `Rata-rata ${item.average.toFixed(1)} °C; rentang ` +
      `${item.min.toFixed(1)}–${item.max.toFixed(1)} °C; ` +
      `${item.count}/${item.expected} sampel (${(item.coverage * 100).toFixed(0)}%); ` +
      `${item.count} pengukuran valid.`;
  }

  if (hourly.length < 2) {
    hotHour.textContent = "Belum cukup data";
    coolHour.textContent = "Belum cukup data";
    hotDetail.textContent =
      "Setiap jam harus memiliki minimal 60 sampel valid dan kelengkapan 75%.";
    coolDetail.textContent =
      "Diperlukan setidaknya dua jam yang memenuhi kriteria tersebut.";
    insight.textContent =
      `Baru ${hourly.length} jam memenuhi syarat. ` +
      "Analisis akan diperbarui ketika data mencukupi.";
    return;
  }

  const highest = Math.max(...hourly.map(item => item.average));
  const lowest = Math.min(...hourly.map(item => item.average));
  const hottest = hourly.filter(item => Math.abs(item.average - highest) < 1e-9);
  const coolest = hourly.filter(item => Math.abs(item.average - lowest) < 1e-9);

  showResult(hotHour, hotDetail, hottest[0]);
  showResult(coolHour, coolDetail, coolest[0]);

  if (Math.abs(highest - lowest) < 1e-9) {
    insight.textContent =
      "Rata-rata suhu semua jam yang memenuhi syarat sama.";
  } else {
    insight.textContent =
      `Selisih rata-rata jam terpanas dan tersejuk ${(highest - lowest).toFixed(1)} °C. ` +
      `${hourly.length} jam memenuhi kriteria minimal 60 sampel dan kelengkapan 75%.`;
  }
}

async function refreshLatest() {
  if (requestInProgress) return;
  requestInProgress = true;

  try {
    const latest = await fetchLatest();

    if (!latest) {
      showOffline("Belum ada data valid dari ESP32.");
      return;
    }

    const age = Date.now() - latest.date.getTime();

    if (age < -5000 || age > OFFLINE_AFTER_MS) {
      showOffline("ESP32 tidak mengirim data terbaru.");
      return;
    }

    showOnline(latest);
  } catch (error) {
    showOffline("Gagal mengambil data dari Supabase.");
    console.error("Kesalahan data terbaru:", error);
  } finally {
    requestInProgress = false;
  }
}

  const { suhu, kelembapan } = req.body;

  if (
    typeof suhu !== "number" ||
    typeof kelembapan !== "number" ||
    !Number.isFinite(suhu) ||
    !Number.isFinite(kelembapan) ||
    suhu < -40 || suhu > 80 ||
    kelembapan < 0 || kelembapan > 100
  ) {
    return res.status(400).json({
      success: false,
      message: "Data sensor tidak valid"
    });
  }

  latest = {
    suhu,
    kelembapan,
    waktu: new Date().toISOString()
  };

  lastReceived = Date.now();
  history.push(latest);

  if (history.length > 500) {
    history.shift();
  }

  res.json({ success: true });
});
